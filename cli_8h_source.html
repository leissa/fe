19/// A small command-line parser for a single command - no subcommands.
20///
21/// Declare the switches with Cli::opt / Cli::arg and bind each one to a variable:
22/// ```
23/// bool show_help = false, verbose = false;
24/// std::string out, in;
25/// std::vector<std::string> plugins;
26///
27/// auto cli = fe::Cli("mim", "The MimIR compiler.")
28/// .help(show_help)
29/// .opt(verbose, "" , "-V", "--verbose", "Be verbose.")
30/// .opt(plugins, "plugin", "-p", "--plugin" , "Loads a plugin; repeatable.")
31/// .grp("Output")
32/// .opt(out , "file" , "-o", "--output" , "Where to write the result.")
33/// .arg(in , "file" , "Input file.");
34///
35/// if (auto err = cli.parse(argc, argv)) throw std::invalid_argument(*err);
36/// if (show_help) std::cout << cli;
37/// ```
38/// A `bool` target - or a callable taking one - is a flag: it takes no hint and is set each time it occurs.
39/// Any other one needs a hint, listed as `<hint>` in the help, and is assigned the value: a `std::string`, an integral
40/// type, a `std::vector` of those, or a callable, which may return a `std::string` to reject the value - a non-empty
41/// one becomes the error of Cli::parse.
42///
43/// The parser understands `--name value`, `--name=value`, `-n value`, `-nvalue`, clustered short flags (`-abc`), and
44/// `--` to end option processing.
45///
46/// Cli::help lays the switches out for a terminal - wrapped to its width and colored via fe::term - whereas
47/// Cli::markdown renders the same information as Markdown tables; Cli::grp splits both into sections.
48class Cli {
49public:
50 Cli() = default;
51 Cli(std::string prog, std::string descr = {})
52 : prog_(std::move(prog))
53 , descr_(std::move(descr)) {}

58 /// An option bound to @p target and named @p sname and/or @p lname - `"-o"` and `"--output"`.
59 template<class T>
60 Cli& opt(T& target, std::string hint = {}, std::string sname = {}, std::string lname = {}, std::string descr = {}) {
61 static_assert(is_flag<T> || std::is_same_v<T, std::string> || Vec<T>::is
62 || is_num<T> || std::is_invocable_v<T&, std::string>,
63 "cannot bind this type to an option");
64 auto& o = opts_.emplace_back(std::move(sname), std::move(lname), std::move(hint), std::move(descr), grp_);
65 assert(is_flag<T> != o.takes_value() && "a flag takes no <hint>, anything else needs one");
66 assert(!(is_flag<T> && o.is_arg()) && "a flag needs a name");
67 o.multi = Vec<T>::is;
68 o.dflt = dflt(target);
69 o.set = [&target](std::string_view s) { return assign(target, s); };
70 return *this;
71 }

77 opt(target, std::move(hint), {}, {}, std::move(descr));

124 /// Applies @p s to @p t; returns a message if @p s does not scan as a `T`.
125 template<class T>
126 static std::string assign(T& t, std::string_view s) {
127 if constexpr (std::is_same_v<T, bool>) {
128 t = true;
129 } else if constexpr (std::is_same_v<T, std::string>) {
130 t = s;
131 } else if constexpr (Vec<T>::is) {
132 typename Vec<T>::Elem elem{};
133 if (auto err = assign(elem, s); !err.empty()) return err;
134 t.emplace_back(std::move(elem));
135 } else if constexpr (is_num<T>) {
136 auto begin = s.data(), end = begin + s.size();
137 if (auto [ptr, ec] = std::from_chars(begin, end, t); ec != std::errc{} || ptr != end)
138 return std::format("'{}' is not a number", s);
139 } else if constexpr (std::is_invocable_r_v<std::string, T&, std::string>) {
140 return t(std::string(s)); // a validating handler reports what it did not like
141 } else if constexpr (std::is_invocable_v<T&, std::string>) {
142 t(std::string(s));
143 } else {
144 t(true); // a flag bound to a callable
145 }
146 return {};
147 }
148
149 /// What @p t holds before parsing and hence shows as `[default: ...]` in the help.
150 template<class T>
151 static std::string dflt(const T& t) {
152 if constexpr (std::is_same_v<T, std::string>)
153 return t;
154 else if constexpr (is_num<T>)
155 return std::format("{}", t);
156 else
157 return {};
158 }
