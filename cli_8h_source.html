19/// A small command-line parser for a single command - no subcommands.
20///
21/// Declare the switches by piping fe::cli::opt / fe::cli::arg into an fe::cli::Cli and bind each one to a variable:
22/// ```
23/// bool show_help = false, verbose = false;
24/// std::string out, in;
25/// std::vector<std::string> plugins;
26///
27/// auto cli = fe::cli::Cli("mim", "The MimIR compiler.")
28/// | fe::cli::help(show_help)
29/// | fe::cli::opt(verbose )["-V"]["--verbose"]("Be verbose.")
30/// | fe::cli::opt(plugins, "plugin" )["-p"]["--plugin" ]("Loads a plugin; repeatable.")
31/// | fe::cli::group("Output")
32/// | fe::cli::opt(out, "file" )["-o"]["--output" ]("Where to write the result.")
33/// | fe::cli::arg(in, "file" ) ("Input file.");
34///
35/// if (auto res = cli.parse(argc, argv); !res) throw std::invalid_argument(res.message());
36/// if (show_help) std::cout << cli;
37/// ```
38/// A target may be a `bool` (for a flag), a `std::string`, an integral type, a `std::vector` of those, or a callable -
39/// invoked with `true` for a flag and with the `std::string` value otherwise.
40/// Such a callable may return a `std::string` to reject that value; a non-empty one becomes the Res::message.
41/// An fe::cli::opt without a hint is a flag; one with a hint takes a value.
42///
43/// The parser understands `--name value`, `--name=value`, `-n value`, `-nvalue`, clustered short flags (`-abc`), and
44/// `--` to end option processing.
45///
46/// Cli::help lays the switches out for a terminal - wrapped to its width and colored via fe::term - whereas Cli::md
47/// renders the same information as Markdown tables; fe::cli::group splits both into sections.

103/// Assigns @p s to @p t; returns a message if @p s does not scan as a `T`.
104template<class T>
105std::string scan(T& t, std::string_view s) {
106 if constexpr (std::is_same_v<T, std::string>) {
107 t = s;
108 } else if constexpr (is_vec<T>) {
109 typename Elem<T>::type elem{};
110 if (auto err = scan(elem, s); !err.empty()) return err;
111 t.emplace_back(std::move(elem));
112 } else if constexpr (is_num<T>) {
113 auto begin = s.data(), end = begin + s.size();
114 if (auto [ptr, ec] = std::from_chars(begin, end, t); ec != std::errc{} || ptr != end)
115 return std::format("'{}' is not a number", s);
116 } else if constexpr (std::is_invocable_r_v<std::string, T&, std::string>) {
117 return t(std::string(s)); // a validating handler reports what it did not like
118 } else if constexpr (std::is_invocable_v<T&, std::string>) {
119 t(std::string(s));
120 } else {
121 static_assert(always_false<T>, "cannot bind this type to an option that takes a value");
122 }
123 return {};
124}
125
126template<class T>
127std::function<std::string()> dflt(T& t) {
128 if constexpr (std::is_same_v<T, std::string>)
129 return [&t] { return t; };
130 else if constexpr (is_num<T>)
131 return [&t] { return std::format("{}", t); };
132 else
133 return {};
134}

138/// One option or positional argument; build one with fe::cli::opt, fe::cli::arg, or fe::cli::help.
139class Opt {
140public:
141 /// Adds @p name - `"-o"` for a short, `"--output"` for a long one.
142 Opt& operator[](std::string name) {
143 names_.emplace_back(std::move(name));
144 return *this;
145 }
146
147 /// Sets the description shown in the help.
148 Opt& operator()(std::string descr) {
149 descr_ = std::move(descr);
150 return *this;
151 }
152
153 /// This Opt must occur at least @p min and at most @p max times.
154 Opt& cardinality(size_t min, size_t max) {
155 min_ = min, max_ = max;
156 return *this;
157 }
158
159private:
160 /// How the names are spelled out in the help - long names align under each other.
161 std::string names() const {
162 std::string shorts, longs;
163 for (const auto& name : names_) {
164 auto& to = name.starts_with("--") ? longs : shorts;
165 if (!to.empty()) to += ", ";
166 to += name;
167 }
168 if (longs.empty()) return shorts;
169 if (shorts.empty()) return " " + longs;
170 return shorts + ", " + longs;
171 }
172
173 size_t width() const {
174 return names_.empty() ? hint_.size() + 2 : names().size() + (value_ ? hint_.size() + 3 : 0);
175 }
176
177 std::string_view label() const { return names_.empty() ? std::string_view(hint_) : names_.back(); }
178
179 std::vector<std::string> names_;
180 std::string hint_, descr_, group_;
181 std::function<std::string(std::string_view)> set_;
182 std::function<std::string()> dflt_;
183 bool value_ = false; ///< Takes a value as opposed to being a flag.
184 bool multi_ = false; ///< Bound to a `std::vector` and hence soaks up any number of values.
185 size_t min_ = 0;
186 size_t max_ = std::numeric_limits<size_t>::max();
187 size_t num_ = 0;
188
189 friend class Cli;
190 template<class T>
191 friend Opt opt(T&);
192 template<class T>
193 friend Opt opt(T&, std::string);
194 template<class T>
195 friend Opt arg(T&, std::string);
196};

198/// A flag: sets @p target to `true` - or invokes it with `true` - each time it occurs.
199template<class T>
200Opt opt(T& target) {
201 Opt o;
202 o.set_ = [&target](std::string_view) -> std::string {
203 if constexpr (std::is_same_v<T, bool>)
204 target = true;
205 else
206 target(true);
207 return {};
208 };
209 return o;
210}

212/// An option that takes a value; @p hint names it in the help as `<hint>`.
213template<class T>
214Opt opt(T& target, std::string hint) {
215 Opt o;
216 o.hint_ = std::move(hint);
217 o.value_ = true;
218 o.multi_ = detail::is_vec<T>;
219 o.set_ = [&target](std::string_view s) { return detail::scan(target, s); };
220 o.dflt_ = detail::dflt(target);
221 return o;
222}

224/// A positional argument; @p hint names it in the help as `<hint>`.
225/// Bind a `std::vector` to soak up all remaining ones.
226template<class T>
227Opt arg(T& target, std::string hint) {
228 auto o = opt(target, std::move(hint));
229 o.dflt_ = {};
230 return o;
231}

325inline Res Cli::parse(int argc, const char* const* argv) {
326 if (prog_.empty() && argc > 0) {
327 prog_ = std::filesystem::path(argv[0]).filename().string();
328 if (prog_.ends_with(".exe")) prog_.resize(prog_.size() - 4);
329 }
330
331 Opt* pos = nullptr;
332 auto next = [&] {
333 for (auto& o : opts_)
334 if (o.names_.empty() && (o.num_ == 0 || o.multi_)) return pos = &o;
335 return pos = nullptr;
336 };
337 next();
338
339 auto set = [](Opt& o, std::string_view name, std::string_view value) -> Res {
340 ++o.num_;
341 if (auto err = o.set_(value); !err.empty()) return std::format("option '{}': {}", name, err);
342 return {};
343 };
344
345 bool only_pos = false;
346 for (int i = 1; i < argc; ++i) {
347 auto s = std::string_view(argv[i]);
348
349 if (!only_pos && s == "--") {
350 only_pos = true;
351 } else if (!only_pos && s.starts_with("--")) {
352 auto eq = s.find('=');
353 auto name = s.substr(0, eq);
354 auto o = find(name);
355 if (!o) return std::format("unknown option '{}'", name);
356
357 if (!o->value_) {
358 if (eq != std::string_view::npos) return std::format("option '{}' does not take a value", name);
359 if (auto res = set(*o, name, {}); !res) return res;
360 } else if (eq != std::string_view::npos) {
361 if (auto res = set(*o, name, s.substr(eq + 1)); !res) return res;
362 } else if (++i < argc) {
363 if (auto res = set(*o, name, argv[i]); !res) return res;
364 } else {
365 return std::format("option '{}' requires a value <{}>", name, o->hint_);
366 }
367 } else if (!only_pos && s.size() > 1 && s.front() == '-') {
368 for (size_t j = 1; j != s.size(); ++j) {
369 char buf[] = {'-', s[j], '\0'};
370 auto name = std::string_view(buf, 2);
371 auto o = find(name);
372 if (!o) return std::format("unknown option '{}'", name);
373
374 if (!o->value_) {
375 if (auto res = set(*o, name, {}); !res) return res;
376 } else if (j + 1 != s.size()) {
377 if (auto res = set(*o, name, s.substr(j + 1)); !res) return res;
378 break;
379 } else if (++i < argc) {
380 if (auto res = set(*o, name, argv[i]); !res) return res;
381 break;
382 } else {
383 return std::format("option '{}' requires a value <{}>", name, o->hint_);
384 }
385 }
386 } else {
387 if (!pos) return std::format("unexpected argument '{}'", s);
388 if (auto res = set(*pos, pos->hint_, s); !res) return res;
389 if (!pos->multi_) next();
390 }
391 }
392
393 for (const auto& o : opts_) {
394 if (o.num_ < o.min_) return std::format("missing option '{}'", o.label());
395 if (o.num_ > o.max_) return std::format("option '{}' must not occur more than {} times", o.label(), o.max_);
396 }
397
398 return {};
399}

401inline void Cli::help(std::ostream& os) const {
402 auto cols = std::clamp<size_t>(term::width(os).value_or(80), 40, 120);
403
404 // Only the specs up to a third of the line dictate the description column; longer ones get a line of their own.
405 size_t spec = 0;
406 auto fits = [&](size_t w) {
407 if (w <= cols / 3) spec = std::max(spec, w);
408 };
409 for (const auto& o : opts_)
410 fits(o.width());
411 for (const auto& s : sections_)
412 for (const auto& [name, descr] : s.rows)
413 fits(name.size());
414 auto col = std::clamp<size_t>(spec + 4, 8, cols / 2);
415
416 auto wrap = [&](std::string_view text) {
417 for (size_t begin = 0, row = 0; begin < text.size(); ++row) {
418 while (begin < text.size() && text[begin] == ' ')
419 ++begin;
420 if (begin >= text.size()) break;
421
422 auto end = std::min(text.size(), begin + cols - col);
423 if (end < text.size())
424 if (auto space = text.rfind(' ', end); space != std::string_view::npos && space > begin) end = space;
425 if (row != 0) os << std::string(col, ' ');
426 os << text.substr(begin, end - begin) << '\n';
427 begin = end;
428 }
429 };
430
431 auto section = [&](std::string_view title) {
432 os << '\n' << term::FG::Yellow << title << ':' << term::FG::Reset << '\n';
433 };
434
435 // Pads what has just been written up to the description column - or breaks the line if it does not fit.
436 auto tail = [&](size_t w, std::string_view descr) {
437 if (w + 2 > col) {
438 os << '\n';
439 w = 0;
440 }
441 os << std::string(col - w, ' ');
442 if (descr.empty())
443 os << '\n';
444 else
445 wrap(descr);
446 };
447
448 auto entry = [&](const Opt& o) {
449 os << " ";
450 if (o.names_.empty()) {
451 os << term::FG::Cyan << '<' << o.hint_ << '>' << term::FG::Reset;
452 } else {
453 os << term::FG::Green << o.names() << term::FG::Reset;
454 if (o.value_) os << ' ' << term::FG::Cyan << '<' << o.hint_ << '>' << term::FG::Reset;
455 }
456
457 auto descr = o.descr_;
458 if (o.dflt_)
459 if (auto d = o.dflt_(); !d.empty()) descr += std::format("{}[default: {}]", descr.empty() ? "" : " ", d);
460 tail(o.width() + 2, descr);
461 };
462
463 os << term::FG::Yellow << "Usage:" << term::FG::Reset << ' ' << usage() << '\n';
464 if (!descr_.empty()) os << '\n' << descr_ << '\n';
465
466 if (std::ranges::any_of(opts_, [](const Opt& o) { return o.names_.empty(); })) {
467 section("Arguments");
468 for (const auto& o : opts_)
469 if (o.names_.empty()) entry(o);
470 }
471
472 for (auto group : groups()) {
473 section(group.empty() ? "Options" : group);
474 for (const auto& o : opts_)
475 if (!o.names_.empty() && o.group_ == group) entry(o);
476 }
477
478 for (const auto& s : sections_) {
479 section(s.title);
480 for (const auto& [name, descr] : s.rows) {
481 os << " " << term::FG::Green << name << term::FG::Reset;
482 tail(name.size() + 2, descr);
483 }
484 }
485
486 if (!epilog_.empty()) os << '\n' << epilog_ << '\n';
487}

489inline void Cli::md(std::ostream& os) const {
490 // A code span is already verbatim - only `|`, which ends the table cell, still has to go.
491 auto esc = [](std::string_view text) {
492 std::string res;
493 bool code = false;
494 for (size_t i = 0, e = text.size(); i != e; ++i) {
495 auto c = text[i];
496 if (c == '`') code = !code;
497 if (code && c != '|') {
498 res += c;
499 continue;
500 }
501 switch (c) {
502 case '&': res += "&amp;"; break;
503 case '<': res += "&lt;"; break;
504 case '>': res += "&gt;"; break;
505 case '|': res += "\\|"; break;
506 // Doxygen turns a bare `--` into an en dash.
507 case '-':
508 if (i + 1 != e && text[i + 1] == '-') res += '\\';
509 res += c;
510 break;
511 default: res += c;
512 }
513 }
514 return res;
515 };
516
517 // A code span still sits in a table cell, so only `|` needs to go.
518 auto code = [](std::string_view text) {
519 std::string res;
520 for (auto c : text) {
521 if (c == '|') res += '\\';
522 res += c;
523 }
524 return res;
525 };
526
527 auto spec = [](const Opt& o) {
528 if (o.names_.empty()) return std::format("<{}>", o.hint_);
529 std::string res;
530 for (const auto& name : o.names_) {
531 if (!res.empty()) res += ", ";
532 res += name;
533 }
534 if (o.value_) res += std::format(" <{}>", o.hint_);
535 return res;
536 };
537
538 auto table = [&](std::string_view title, std::string_view head, auto&& pred) {
539 os << std::format("\n### {}\n\n| {} | Description |\n| --- | --- |\n", title, head);
540 for (const auto& o : opts_) {
541 if (!pred(o)) continue;
542 auto descr = esc(o.descr_);
543 if (o.dflt_)
544 if (auto d = o.dflt_(); !d.empty())
545 descr += std::format("{}[default: `{}`]", descr.empty() ? "" : " ", code(d));
546 os << std::format("| `{}` | {} |\n", code(spec(o)), descr);
547 }
548 };
549
550 os << std::format("```\n{}\n```\n", usage());
551 if (!descr_.empty()) os << '\n' << esc(descr_) << '\n';
552
553 if (std::ranges::any_of(opts_, [](const Opt& o) { return o.names_.empty(); }))
554 table("Arguments", "Argument", [](const Opt& o) { return o.names_.empty(); });
555
556 for (auto group : groups())
557 table(group.empty() ? "Options" : group, "Option",
558 [&](const Opt& o) { return !o.names_.empty() && o.group_ == group; });
559
560 for (const auto& s : sections_) {
561 if (s.rows.empty()) { // a Section without rows only groups the ones below it, hence one level up
562 os << std::format("\n## {}\n", esc(s.title));
563 continue;
564 }
565 os << std::format("\n### {}\n\n| {} | Description |\n| --- | --- |\n", esc(s.title), s.head);
566 for (const auto& [name, descr] : s.rows)
567 os << std::format("| `{}` | {} |\n", code(name), esc(descr));
568 }
569
570 if (!epilog_.empty()) os << '\n' << esc(epilog_) << '\n';
571}
