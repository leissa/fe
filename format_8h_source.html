21/// Wrap a callable `f(std::ostream&) -> std::ostream&` so it streams via `operator<<` and `std::format`.
22/// Useful for inline ad-hoc formatting:
23/// ```
24/// auto greet = fe::StreamFn{[](std::ostream& os) { os << "hi"; }};
25/// std::cout << greet;
26/// std::format("{}", greet);
27/// ```
28template<class F>
29requires std::invocable<const F&, std::ostream&>
30class StreamFn {
31public:
32 constexpr StreamFn(F f)
33 : f_(std::move(f)) {}
34
35 friend std::ostream& operator<<(std::ostream& os, StreamFn const& s) {
36 if constexpr (std::same_as<std::invoke_result_t<F const&, std::ostream&>, std::ostream&>)
37 return std::invoke(s.f_, os);
38 else
39 return std::invoke(s.f_, os), os;
40 }
41
42private:
43 F f_;
44};
