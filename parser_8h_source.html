11/// The blueprint for a [recursive descent](https://en.wikipedia.org/wiki/Recursive_descent_parser)/
12/// [ascent parser](https://en.wikipedia.org/wiki/Recursive_ascent_parser) using a @p K lookahead of `Tok`ens.
13/// Parser::accept and Parser::expect indicate failure by constructing a @p Tok%en with its default constructor.
14/// Hence, @p Tok must be default-constructible *and* testable as a `bool` (to check for that failure):
15/// ```
16/// class Tok {
17/// public:
18/// enum class Tag {
19/// Nil,
20/// // ...
21/// };
22/// Tok() {} // default constructor yields the "failure" token
23/// // ...
24/// explicit operator bool() const { return tag_ != Tag::Nil; }
25/// // ...
26/// };
27///
28/// // Your Parser:
29/// if (auto tok = accept(Tok::Tag::My_Tag)) {
30/// do_something(tok);
31/// }
32/// ```
33template<class Tok, class Tag, size_t K, class S>
34requires std::is_default_constructible_v<Tok>
35 && (std::is_convertible_v<Tok, bool> || std::is_constructible_v<bool, Tok>)

37private:
38 S& self() { return *static_cast<S*>(this); }

41protected:

43 ///@{
44 void init() {
45 ahead_.reset();
46 for (size_t i = 0; i != K; ++i)
47 ahead_[i] = self().lexer().lex();
48 curr_ = ahead().loc().anew_begin();
