17/// What fe::Parser needs of its Tok%en type: a Tag and a Loc, copied in and out of the lookahead by value.
18/// Parser::accept and Parser::expect indicate failure by returning a default-constructed Tok%en,
19/// so `Tok()` must be falsy and tagged Tag::Nil.
20/// That sentinel is only checkable here if `Tok()` is usable in a constant expression - Parser::init asserts it.
21template<class Tok, class Tag>
22concept Token = requires(const Tok tok) {
23 Tag::Nil;
24 { tok.tag() } -> std::convertible_to<Tag>;
25 { tok.loc() } -> std::convertible_to<Loc>;
26 requires std::equality_comparable<Tag>;
27 requires std::semiregular<Tok>;
28 requires std::constructible_from<bool, Tok>;
29 requires !requires { std::integral_constant<Tag, Tok().tag()>(); } || Tok().tag() == Tag::Nil;
30};

33template<class S, class Tok>
34concept Lexable = requires(S& s) {
35 { s.lexer().lex() } -> std::convertible_to<Tok>;

38/// The blueprint for a [recursive descent](https://en.wikipedia.org/wiki/Recursive_descent_parser)/
39/// [ascent parser](https://en.wikipedia.org/wiki/Recursive_ascent_parser) using a @p K lookahead of `Tok`ens.
40/// Parser::accept and Parser::expect indicate failure by constructing a @p Tok%en with its default constructor.
41/// Hence, @p Tok must be an fe::Token - default-constructible, testable as a `bool`, and tagged Tag::Nil:
42/// ```
43/// class Tok {
44/// public:
45/// enum class Tag {
46/// Nil,
47/// // ...
48/// };
49/// Tok() {} // default constructor yields the "failure" token
50/// // ...
51/// explicit operator bool() const { return tag_ != Tag::Nil; }
52/// // ...
53/// };
54///
55/// // Your Parser:
56/// if (auto tok = accept(Tok::Tag::My_Tag)) {
57/// do_something(tok);
58/// }
59/// ```
60/// @p S must provide the Lexer to pull from and the Driver to report to - publicly,
61/// so that fe::Lexable and fe::Diagnosable hold outside this class too:
62/// ```
63/// class MyParser : public fe::Parser<Tok, Tok::Tag, K, MyParser> {
64/// public:
65/// Lexer& lexer(); ///< Parser::lex pulls the next Tok%en from here.
66/// fe::Driver& driver(); ///< The default diagnostics below land in its Driver::error.
67/// };
68/// ```
69/// Parser::syntax_err and Parser::unanchored_err come with a default; declare either in @p S to word it differently.
70/// Each is one name with one signature, so a declaration in @p S replaces it outright - no `using` of the base needed.
71template<class Tok, class Tag, size_t K, class S>
72requires Token<Tok, Tag> && (K >= 1) class Parser {
73private:
74 S& self() { return *static_cast<S*>(this); }
75 const S& self() const { return *static_cast<const S*>(this); }
76
77protected:
78 /// @name Construction
79 ///@{
80 void init() {
81 static_assert(Lexable<S, Tok>, "provide `Lexer& lexer()` in your parser - its `lex` is what feeds the Parser");
82 assert(!Tok() && Tok().tag() == Tag::Nil && "a default-constructed Tok must be the Nil token");
83 ahead_.reset();
84 for (size_t i = 0; i != K; ++i)
85 ahead_[i] = self().lexer().lex();
86 curr_ = ahead().loc().anew_begin();
87 }
88 ///@}

136 /// If Parser::ahead() is a @p tag, consume and return it, otherwise yield `std::nullopt`.
137 Tok accept(Tag tag) {
138 if (tag != ahead().tag()) return {};

142 /// Parser::lex Parser::ahead() which must be a @p tag.
143 /// Issue error with @p ctxt otherwise.
144 Tok expect(Tag tag, Cite ctxt) {
145 if (ahead().tag() == tag) return lex();

188 /// @name Anchor
189 /// An *anchor* is a @p Tag that an enclosing context is waiting for.
190 /// E.g., while parsing a parenthesized expression, `)` is anchored:
191 /// a nested parser must not swallow it but bail out, so the enclosing context can Parser::expect it.
192 /// A `)` that is *not* anchored, however, is simply bogus and Parser::recover discards it.
193 ///@{
194
195 /// Factory method to build a Parser::ScopedAnchor; Parser::expect @p r_tag yourself at the end of the scope.
196 /// Hand the Tok%en that opened this context to @p l_tok and Parser::syntax_err notes it on a missing @p r_tag,
197 /// which is all it takes to point a `)` that never came back at its `(`.
198 /// Use like this:
199 /// ```
200 /// if (auto paren_l = accept(Tag::D_paren_l)) {
201 /// auto _ = anchor(Tag::D_paren_r, paren_l);
202 /// auto expr = parse_expr();
203 /// expect(Tag::D_paren_r, "parenthesized expression");
204 /// return expr;
205 /// }
206 /// ```
207 [[nodiscard]] ScopedAnchor anchor(Tok l_tok, Tag r_tag) { return {*this, l_tok, r_tag}; }

209 /// The innermost Anchor waiting for @p r_tag - `nullptr` if no enclosing context is.
210 /// Scans the innermost anchor first, but *any* enclosing context counts.
211 const Anchor* find_anchor(Tag r_tag) const {
212 if (r_tag == Tag::Nil) return nullptr;
213 auto i
214 = std::find_if(anchors_.rbegin(), anchors_.rend(), [r_tag](const Anchor& a) { return a.r_tag == r_tag; });
215 return i == anchors_.rend() ? nullptr : &*i;
216 }

218 /// Is @p r_tag anchored by an enclosing context?
219 bool is_anchored(Tag r_tag) const { return find_anchor(r_tag); }

221 /// Parser::lex all Tok%ens whose Tag satisfies @p pred and that are not Parser::is_anchored;
222 /// report the whole run as a single `S::unanchored_err`.
223 /// This turns an otherwise fatal Tok%en into a mere error message and keeps the current parser going.
224 /// One mistake discards one run, so one run is one diagnostic: a message per token buries the real error.
225 template<std::predicate<Tag> P>
226 void recover(P pred, Cite ctxt) {
227 auto discard = [this, &pred] { return pred(ahead().tag()) && !is_anchored(ahead().tag()); };
228 if (!discard()) return;
229
230 auto first = lex();
231 auto loc = first.loc();
232 size_t n = 1;
233 for (; discard(); ++n)
234 loc.end = lex().loc().end;
235
236 self().unanchored_err(first, loc, n, ctxt);
237 }
238
239 /// As above but only recovers from @p tag.
240 void recover(Tag tag, Cite ctxt) {
241 recover([tag](Tag t) { return t == tag; }, ctxt);
242 }

271 /// Parser::expect did not find @p what while parsing @p ctxt; @p got defaults to Parser::ahead.
272 /// @p ctxt is markup, so backtick a literal token within it yourself.
273 /// A @p what that an enclosing context anchors gets a Note pointing back at the Tok%en that opened it -
274 /// the `(` of a `)` that never came - as long as Parser::anchor was handed that token.
275 fe::Error& syntax_err(Expected what, Cite ctxt, Tok got = {}) {
276 static_assert(Diagnosable<S>, "provide `fe::Driver& driver()` in your parser - or a `syntax_err` of your own");
277 static_assert(Formattable<Tok>, "provide a `std::formatter` for your Tok - or a `syntax_err` of your own");
278 if (!got) got = ahead();
279 auto& err = error().e(got.loc(), "expected {}, got `{}` while parsing {}", what.what, got, ctxt);
280 if (auto* a = find_anchor(what.tag); a && a->l_tok)
281 err.n(a->l_tok.loc(), "unmatched `{}` opened here", a->l_tok);
282 return err;
283 }

286 fe::Error& unanchored_err(Tok tok, Loc loc, size_t n, Cite ctxt) {
287 static_assert(Diagnosable<S>,
288 "provide `fe::Driver& driver()` in your parser - or an `unanchored_err` of your own");
289 static_assert(Formattable<Tok>, "provide a `std::formatter` for your Tok - or an `unanchored_err` of your own");
290 if (n == 1) return error().e(loc, "ignoring unmatched `{}` while parsing {}", tok, ctxt);
291 return error().e(loc, "ignoring {} unmatched tokens starting with `{}` while parsing {}", n, tok, ctxt);
292 }
