14/// The blueprint for a [recursive descent](https://en.wikipedia.org/wiki/Recursive_descent_parser)/
15/// [ascent parser](https://en.wikipedia.org/wiki/Recursive_ascent_parser) using a @p K lookahead of `Tok`ens.
16/// Parser::accept and Parser::expect indicate failure by constructing a @p Tok%en with its default constructor.
17/// Hence, @p Tok must be default-constructible *and* testable as a `bool` (to check for that failure):
18/// ```
19/// class Tok {
20/// public:
21/// enum class Tag {
22/// Nil,
23/// // ...
24/// };
25/// Tok() {} // default constructor yields the "failure" token
26/// // ...
27/// explicit operator bool() const { return tag_ != Tag::Nil; }
28/// // ...
29/// };
30///
31/// // Your Parser:
32/// if (auto tok = accept(Tok::Tag::My_Tag)) {
33/// do_something(tok);
34/// }
35/// ```
36/// The Parser itself does not report anything; @p S must provide the Lexer to pull from and the diagnostics:
37/// ```
38/// class MyParser : public fe::Parser<Tok, Tok::Tag, K, MyParser> {
39/// Lexer& lexer(); ///< Parser::lex pulls the next Tok%en from here.
40/// void syntax_err(Tag, std::string_view ctxt); ///< Parser::expect did not find its Tag.
41/// void unanchored_err(Tok, std::string_view ctxt); ///< Parser::recover discarded this Tok%en.
42///
43/// friend fe::Parser<Tok, Tok::Tag, K, MyParser>; ///< Otherwise, these may be private.
44/// };
45/// ```
46template<class Tok, class Tag, size_t K, class S>
47requires std::is_default_constructible_v<Tok>
48 && (std::is_convertible_v<Tok, bool> || std::is_constructible_v<bool, Tok>)
49class Parser {
50private:

58 ahead_.reset();

60 ahead_[i] = self().lexer().lex();

138 /// @name Anchor
139 /// An *anchor* is a @p Tag that an enclosing context is waiting for.
140 /// E.g., while parsing a parenthesized expression, `)` is anchored:
141 /// a nested parser must not swallow it but bail out, so the enclosing context can Parser::expect it.

166 /// auto _ = this->anchor(Tag::D_paren_r);

178 /// Parser::lex all Tok%ens whose Tag satisfies @p pred and that are not Parser::anchored;
179 /// report each one as `S::unanchored_err`.

181 template<std::predicate<Tag> P>
182 void recover(P pred, std::string_view ctxt) {
183 while (pred(ahead().tag()) && !anchored(ahead().tag()))
184 self().unanchored_err(lex(), ctxt);
185 }
