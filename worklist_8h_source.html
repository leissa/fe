15/// A FIFO queue over a Vector.
16/// Popped elements stay in place until they make up half of the storage, so `pop` is amortized constant.
17template<class T, size_t N = Default_Inlined_Size<T>>
18class VectorQueue {
19public:
20 using value_type = T;
21
22 bool empty() const { return head_ == vec_.size(); }
23 size_t size() const { return vec_.size() - head_; }
24
25 T& front() { return vec_[head_]; }
26 const T& front() const { return vec_[head_]; }
27 T& back() { return vec_.back(); }
28 const T& back() const { return vec_.back(); }
29
30 void push(const T& val) { vec_.push_back(val); }
31 void push(T&& val) { vec_.push_back(std::move(val)); }
32 template<class... Args>
33 decltype(auto) emplace(Args&&... args) {
34 return vec_.emplace_back(std::forward<Args>(args)...);
35 }
36
37 void pop() {
38 if (++head_ == vec_.size()) {
39 vec_.clear();
40 head_ = 0;
41 } else if (head_ >= 2 * N && 2 * head_ >= vec_.size()) {
42 vec_.erase(vec_.begin(), vec_.begin() + head_);
43 head_ = 0;
44 }
45 }
46
47private:
48 Vector<T, N> vec_;
49 size_t head_ = 0;
50};
