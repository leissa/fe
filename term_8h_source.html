14/// Lightweight stream-based terminal colors for diagnostics and CLI output.
15///
16/// Include `fe/term.h` and stream a @ref fe::term::FG value into an `std::ostream`:
17/// ```
18/// std::cerr << fe::term::FG::Red << "error: " << fe::term::FG::Reset << "unexpected token\n";
19/// ```
20///
21/// The current behavior is controlled via @ref fe::term::Mode and can be overridden with
22/// @ref fe::term::set_mode. In @ref fe::term::Mode::Auto, colors are emitted only for
23/// `std::cout`, `std::cerr`, `std::clog`, or streams sharing those buffers when they refer to
24/// terminals. FE also respects the common environment conventions `NO_COLOR`, `CLICOLOR=0`, and
25/// `CLICOLOR_FORCE` (unless it is set to `0`).
26///
27/// Note that a `std::formatter` never sees the destination stream - it formats into a detached buffer.
28/// Hence, embedding a @ref fe::term::FG value in a `std::format`/`std::print` format string resolves
29/// @ref fe::term::Mode::Auto to "no color".
30/// If you emit colors this way, call @ref fe::term::resolve_mode once at startup to decide
31/// @ref fe::term::Mode::Auto up front based on a representative stream.
32///
33/// The mode and @ref fe::term::auto_detached live in the fe library rather than in this header, so a
34/// shared library loaded via fe::dl sees whatever the host set instead of starting over from the defaults.
35///
36/// Use @ref fe::term::use_color to branch on whether color will actually be emitted, e.g. to keep a
37/// plain-text fallback in sync with the colored rendering.

92/// Does @p str spell an escape - `` \` `` or `\\` - at position @p i of `[i, end)`?
93constexpr bool escape(std::string_view str, size_t i, size_t end) noexcept {
94 return str[i] == '\\' && i + 1 != end && (str[i + 1] == '`' || str[i + 1] == '\\');
95}

106/// Splits @p str into its `` `citation` `` markup and invokes `f(begin, end, cited)` on each piece;
107/// an unpaired backtick is no citation. This is the one place that knows the grammar.
108template<class F>
109void scan_cite(std::string_view str, F&& f) {
110 for (size_t i = 0, e = str.size(); i != e;) {
111 auto l = tick(str, i);
112 auto r = l == std::string_view::npos ? l : tick(str, l + 1);
113 if (r == std::string_view::npos) return f(i, e, false);
114
115 f(i, l, false);
116 f(l + 1, r, true);
117 i = r + 1;
118 }
119}

166/// Escapes @p str into @p out so that render_cite reproduces it verbatim instead of reading it as markup.
167/// Escaping the backslashes as well keeps a trailing one from swallowing the backtick that follows it.
168template<class O>
169O escape_cite_to(O out, std::string_view str) {
170 for (auto c : str) {
171 if (c == '`' || c == '\\') *out++ = '\\';
172 *out++ = c;
173 }
174 return out;
175}

229/// std::vformat, but each argument renders as data instead of as markup; see Cite for the exception.
230/// @note The wrappers are lambda *parameters* because `std::make_format_args` does not bind rvalues.
231template<class... Args>
232std::string vformat_cite(std::string_view fmt, const Args&... args) {
233 auto vformat = [fmt]<class... A>(A... a) { return std::vformat(fmt, std::make_format_args(a...)); };
234 return vformat(cite_arg_t<Args>{args}...);
235}
