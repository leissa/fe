33/// Lightweight stream-based terminal colors for diagnostics and CLI output.
34///
35/// Include `fe/term.h` and stream a @ref fe::term::FG value into an `std::ostream`:
36/// ```
37/// std::cerr << fe::term::FG::Red << "error: " << fe::term::FG::Reset << "unexpected token\n";
38/// ```
39///
40/// The current behavior is controlled via @ref fe::term::Mode and can be overridden with
41/// @ref fe::term::set_mode. In @ref fe::term::Mode::Auto, colors are emitted only for
42/// `std::cout`, `std::cerr`, `std::clog`, or streams sharing those buffers when they refer to
43/// terminals. FE also respects the common environment conventions `NO_COLOR`, `CLICOLOR=0`, and
44/// `CLICOLOR_FORCE` (unless it is set to `0`).
45///
46/// Note that a `std::formatter` never sees the destination stream - it formats into a detached buffer.
47/// Hence, embedding a @ref fe::term::FG value in a `std::format`/`std::print` format string resolves
48/// @ref fe::term::Mode::Auto to "no color".
49/// If you emit colors this way, call @ref fe::term::resolve_mode once at startup to decide
50/// @ref fe::term::Mode::Auto up front based on a representative stream.
51///
52/// Use @ref fe::term::use_color to branch on whether color will actually be emitted, e.g. to keep a
53/// plain-text fallback in sync with the colored rendering.
54namespace fe::term {

190/// Does @p str spell an escape - `` \` `` or `\\` - at position @p i of `[i, end)`?
191constexpr bool escape(std::string_view str, size_t i, size_t end) noexcept {
192 return str[i] == '\\' && i + 1 != end && (str[i + 1] == '`' || str[i + 1] == '\\');
193}
194
195/// Index of the next backtick at or after @p i that is not escaped, or `npos`.
196inline size_t tick(std::string_view str, size_t i) noexcept {
197 for (; i != str.size(); ++i)
198 if (escape(str, i, str.size()))
199 ++i;
200 else if (str[i] == '`')
201 return i;
202 return std::string_view::npos;
203}
204
205/// Streams `[begin, end)` of @p str, dropping the leading backslash of every escape.
206inline void stream_raw(std::ostream& os, std::string_view str, size_t begin, size_t end) {
207 for (auto i = begin; i != end; ++i) {
208 if (escape(str, i, end)) ++i;
209 os << str[i];
210 }
211}
212
213/// Columns `[begin, end)` of @p str occupies once streamed via stream_raw - one less per escape.
214inline size_t raw_width(std::string_view str, size_t begin, size_t end) noexcept {
215 size_t width = 0;
216 for (auto i = begin; i != end; ++i, ++width)
217 if (escape(str, i, end)) ++i;
218 return width;
219}

233/// Whether color escape sequences are emitted for @p os right now.
234/// In Mode::Auto this is decided by whether @p os refers to a terminal, while a detached buffer
235/// (anything a `std::formatter` writes into) follows @ref resolve_mode.
236/// Use this to keep a plain-text fallback in sync with what @ref operator<<(std::ostream&, FG) will emit,
237/// e.g. to spell out a marker only when it cannot be conveyed by color.
238inline bool use_color(std::ostream& os) noexcept {
239 auto s = detail::stream(os);
240 // clang-format off
241 switch (mode()) {
242 case Mode::Always: return true;
243 case Mode::Never: return false;
244 case Mode::Auto: return s == detail::Stream::Unknown ? auto_detached() : detail::is_terminal(s);
245 default: fe::unreachable();
246 }
247 // clang-format on
248}

250/// Number of columns of the terminal @p os refers to.
251/// Unlike @ref use_color, this ignores @ref Mode and always asks the actual stream.
252/// @returns `std::nullopt` if @p os is not a terminal or its size cannot be determined.
253inline std::optional<size_t> width(std::ostream& os) noexcept {
254 auto s = detail::stream(os);
255 if (!detail::is_terminal(s)) return {};
256
257#ifdef _WIN32
258 auto handle = GetStdHandle(s == detail::Stream::Stdout ? STD_OUTPUT_HANDLE : STD_ERROR_HANDLE);
259 if (CONSOLE_SCREEN_BUFFER_INFO info; GetConsoleScreenBufferInfo(handle, &info)) {
260 if (auto cols = info.srWindow.Right - info.srWindow.Left + 1; cols > 0) return size_t(cols);
261 }
262#else
263 auto fd = s == detail::Stream::Stdout ? STDOUT_FILENO : STDERR_FILENO;
264 if (winsize ws; ::ioctl(fd, TIOCGWINSZ, &ws) == 0 && ws.ws_col > 0) return size_t(ws.ws_col);
265#endif
266 return {};
267}

276/// Decides Mode::Auto for detached buffers, depending on whether @p os refers to a terminal.
277/// A `std::formatter` cannot see its destination stream, so FG values embedded in a
278/// `std::format`/`std::print` format string never detect a terminal on their own.
279/// Call this once at startup to make formatted output colored as well:
280/// ```
281/// fe::term::resolve_mode(); // decide based on stderr
282/// std::print(std::cerr, "{}error:{} ...", fe::term::FG::Red, fe::term::FG::Reset);
283/// ```
284/// A real stream still decides on its own, and explicit modes are left untouched.
285inline void resolve_mode(std::ostream& os = std::cerr) noexcept {
286 set_auto_detached(detail::is_terminal(detail::stream(os)));
287}

298/// Escapes @p str into @p out so that render_cite reproduces it verbatim instead of reading it as markup.
299/// Escaping the backslashes as well keeps a trailing one from swallowing the backtick that follows it.
300template<class O>
301O escape_cite_to(O out, std::string_view str) {
302 for (auto c : str) {
303 if (c == '`' || c == '\\') *out++ = '\\';
304 *out++ = c;
305 }
306 return out;
307}
308
309/// As above but into a fresh `std::string`.
310inline std::string escape_cite(std::string_view str) {
311 auto res = std::string();
312 res.reserve(str.size());
313 escape_cite_to(std::back_inserter(res), str);
314 return res;
315}

331/// A literal and a Cited convert implicitly - both are text someone wrote as markup.

333/// A `const char*` - a ternary of two literals, say - counts as runtime text and needs `Cite(s)` too.

346 constexpr explicit operator bool() const noexcept { return !str.empty(); } ///< Is not empty?

383/// Streams @p str into @p os, coloring each `` `citation` `` and dropping its backticks - or keeping them
384/// verbatim without color; `` \` `` is a literal backtick and `\\` a literal backslash. This is the convention
385/// fe::CodeDiag renders a diagnostic message with; use it to apply the same convention elsewhere, e.g.
386/// fe::Cli::help.
387inline void render_cite(std::ostream& os, std::string_view str, bool color) {
388 for (size_t i = 0, e = str.size(); i != e;) {
389 auto l = detail::tick(str, i);
390 auto r = l == std::string_view::npos ? l : detail::tick(str, l + 1);
391 if (r == std::string_view::npos) { // unpaired: not a citation
392 detail::stream_raw(os, str, i, e);
393 break;
394 }
395
396 detail::stream_raw(os, str, i, l);
397 if (color)
398 os << FG::Cyan;
399 else
400 os << '`';
401 detail::stream_raw(os, str, l + 1, r);
402 if (color)
403 os << FG::Reset;
404 else
405 os << '`';
406 i = r + 1;
407 }
408}

410/// As above but lets @p os decide the coloring; mirrors cite_width.
411inline void render_cite(std::ostream& os, std::string_view str) { render_cite(os, str, use_color(os)); }

413/// Number of columns @p str actually occupies once render_cite renders it with @p color - fewer than
414/// `str.size()` by the backticks/backslashes render_cite drops.
415inline size_t cite_width(std::string_view str, bool color) {
416 size_t width = 0;
417
418 for (size_t i = 0, e = str.size(); i != e;) {
419 auto l = detail::tick(str, i);
420 auto r = l == std::string_view::npos ? l : detail::tick(str, l + 1);
421 if (r == std::string_view::npos) { // unpaired: not a citation
422 width += detail::raw_width(str, i, e);
423 break;
424 }
425
426 width += detail::raw_width(str, i, l);
427 width += color ? 0 : 1;
428 width += detail::raw_width(str, l + 1, r);
429 width += color ? 0 : 1;
430 i = r + 1;
431 }
432
433 return width;
434}
