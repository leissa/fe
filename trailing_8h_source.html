15/// CRTP base that gives @p Self one array per element type in `Self::Trail_Types` directly behind it
16/// in memory - the C flexible array member, generalized to several arrays of different type.
17/// @p Self must be created through Arena::mk or Arena::ref, which allocate the arrays along with
18/// @p Self and fill them from the *last* `Self::Trail_Types` arguments, in that order.
19/// This base is empty - the element counts live at the front of the trailing block - so that
20/// `Self::Trail_Types` may name types only declared inside @p Self.
21/// ```
22/// class Interval : public Node, public fe::Trailing<Interval> {
23/// public:
24/// using Trail_Types = std::tuple<const Expr*, const Expr*>;
25/// Interval(Loc loc, Tag from, Tag to); // no parameter for either array
26/// auto from_args() const { return trail<0>(); } // fe::View<const Expr*>
27/// auto to_args() const { return trail<1>(); }
28/// };
29///
30/// arena.ref<Interval>(loc, from, to, from_args, to_args);
31/// ```
32template<class Self>

34public:
