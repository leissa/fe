24/// Streams @p str, rendering the `` `code` `` spans a diagnostic cites in color.
25/// Color and backticks are alternative ways of setting a citation apart, so the backticks are dropped
26/// when coloring and kept verbatim otherwise.
27inline std::ostream& stream_code(std::ostream& os, std::string_view str) {
28 if (!term::use_color(os)) return os << str;
29
30 for (size_t i = 0, e = str.size(); i != e;) {
31 auto l = str.find('`', i);
32 if (l == std::string_view::npos) return os << str.substr(i);
33 auto r = str.find('`', l + 1);
34 if (r == std::string_view::npos) return os << str.substr(i); // unpaired: not a citation
35 os << str.substr(i, l - i) << term::FG::Cyan << str.substr(l + 1, r - l - 1) << term::FG::Reset;
36 i = r + 1;
37 }
38 return os;
39}

222 void msg_(Loc loc, Tag tag, const std::function<std::string()>& fmt) {
223 assert(tag != Tag::Note && "a note belongs to Error::note");
224 auto d = diag();
225 if (tag == Tag::Warn && d.werror) tag = Tag::Error;
226
227 if (tag == Tag::Error && d.max_errors != 0 && num_errors() >= d.max_errors) {
228 truncated_ = dropped_ = true;
229 return;
230 }
231
232 dropped_ = false;
233 ++num_[size_t(tag)];
234 msgs_.emplace_back(loc, tag, render_(fmt));
235 }

237 void note_(Loc loc, const std::function<std::string()>& fmt) {
238 if (dropped_) return;
239 assert(!msgs_.empty() && "a note needs an error or warning to attach to");
240 ++num_[size_t(Tag::Note)];
241 msgs_.back().notes.emplace_back(loc, render_(fmt));
242 }

244 /// Streamed piecewise instead of via std::format: a std::formatter cannot see its destination stream,
245 /// so embedded term::FG values would resolve Mode::Auto to "no color"; see fe/term.h.
246 std::ostream& header_(std::ostream& os, Loc loc, Tag tag, std::string_view str) const {
247 os << term::FG::Yellow << loc << ": " << tag << ": " << term::FG::Reset;
248 return stream_code(os, str);

251 std::ostream& stream(std::ostream& os) const {
252 auto d = diag();
253 auto snippet = [&](Loc loc, Tag tag) {
254 if (!d.no_snippet) os << Snippet{loc, tag2color(tag), d.gutter, d.max_rows};
255 };
256
257 for (const auto& msg : msgs_) {
258 header_(os, msg.loc, msg.tag, msg.str) << '\n';
259 snippet(msg.loc, msg.tag);
260
261 for (const auto& note : msg.notes) {
262 if (note.loc) {
263 os << std::format("{:>{}} ", "", d.gutter);
264 header_(os, note.loc, Tag::Note, note.str) << '\n';
265 snippet(note.loc, Tag::Note);
266 } else {
267 os << term::FG::Gray << std::format("{:>{}} = ", "", d.gutter) << Tag::Note << ": "
268 << term::FG::Reset;
269 stream_code(os, note.str) << '\n';
270 }
271 }
272 }
273
274 return os;
275 }

277 /// The `n error(s), m warning(s) encountered` line that closes a dump.
278 std::ostream& summary(std::ostream& os) const {
279 if (empty()) return os;
280
281 auto sep = std::string_view();
282 if (auto n = num_errors()) {
283 os << sep << n << " error(s)";
284 sep = ", ";
285 }
286 if (auto n = num_warnings()) os << sep << n << " warning(s)";
287 os << " encountered";
288 if (truncated_) os << "; further diagnostics dropped";
289 return os << '\n';
290 }
