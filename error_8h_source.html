22/// Streams @p str, rendering the `` `code` `` spans a diagnostic cites in color.
23/// Color and backticks are alternative ways of setting a citation apart, so the backticks are dropped
24/// when coloring and kept verbatim otherwise.
25inline std::ostream& stream_code(std::ostream& os, std::string_view str) {
26 if (!term::use_color(os)) return os << str;
27
28 for (size_t i = 0, e = str.size(); i != e;) {
29 auto l = str.find('`', i);
30 if (l == std::string_view::npos) return os << str.substr(i);
31 auto r = str.find('`', l + 1);
32 if (r == std::string_view::npos) return os << str.substr(i); // unpaired: not a citation
33 os << str.substr(i, l - i) << term::FG::Cyan << str.substr(l + 1, r - l - 1) << term::FG::Reset;
34 i = r + 1;
35 }
36 return os;
37}

95 /// A @p loc overlapping the primary one is already covered by its snippet and so points nowhere new.

147 /// Renders each Tag::Error/Tag::Warn with its source snippet and its Tag::Note%s underneath.
148 /// A Tag::Note pointing at a Loc of its own reads as a diagnostic of its own - header line plus snippet;
149 /// one about the primary Loc has nowhere else to point and stays a `= note:` continuation line.
150 friend std::ostream& operator<<(std::ostream& os, const Error& e) { return e.stream(os); }

155 /// Loc of the Tag::Error/Tag::Warn that subsequent Tag::Note%s belong to.
156 Loc primary_loc() const {
157 for (auto i = msgs_.rbegin(), e = msgs_.rend(); i != e; ++i)
158 if (i->tag != Tag::Note) return i->loc;
159 return {};

162 /// Streamed piecewise instead of via std::format: a std::formatter cannot see its destination stream,
163 /// so embedded term::FG values would resolve Mode::Auto to "no color"; see fe/term.h.
164 std::ostream& header(std::ostream& os, const Msg& msg) const {
165 os << term::FG::Yellow << msg.loc << ": " << msg.tag << ": " << term::FG::Reset;
166 return stream_code(os, msg.str);
167 }
168
169 std::ostream& stream(std::ostream& os) const {
170 auto [gutter, max_rows, no_snippet] = diag();
171 auto primary = Loc();
172
173 auto snippet = [&](const Msg& msg) {
174 if (!no_snippet) os << Snippet{msg.loc, tag2color(msg.tag), gutter, max_rows};
175 };
176
177 for (const auto& msg : msgs_) {
178 if (msg.tag == Tag::Note) {
179 // A note pointing elsewhere reads as a diagnostic of its own; one about the primary Loc has no
180 // other place to name and stays a continuation line.
181 if (msg.loc && msg.loc != primary) {
182 os << std::format("{:>{}} ", "", gutter);
183 header(os, msg) << '\n';
184 snippet(msg);
185 } else {
186 os << term::FG::Gray << std::format("{:>{}} = ", "", gutter) << msg.tag << ": " << term::FG::Reset;
187 stream_code(os, msg.str) << '\n';
188 }
189 } else {
190 primary = msg.loc;
191 header(os, msg) << '\n';
192 snippet(msg);
193 }
194 }
195
196 return os.flush();
197 }
