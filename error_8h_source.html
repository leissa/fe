117 /// A Note that points *elsewhere*; dropped when @p loc adds nothing.
118 /// A @p loc overlapping the primary one is already covered by its snippet and so points nowhere new.
119 /// An invalid @p loc points *nowhere* and degrades to the `= note:` continuation above.
120 /// The renderer gives @p loc a header line of its own, so phrase the message to stand alone.
121 template<class... Args> Error& n(Loc loc, cite_string<Args...> s, Args&&... args) {
122 if (loc && (loc & primary_loc_())) return *this;
123 note_(loc, [&] { return term::detail::vformat_cite(s.get(), args...); });
124 return *this;

169 /// Hands every Msg, its Note%s, and the closing summary to Diag.
170 friend std::ostream& operator<<(std::ostream& os, const Error& e) {
171 const auto& diag = e.diag();
172
173 for (const auto& msg : e.msgs_) {
174 diag.header(os, msg.loc, msg.tag, msg.str);
175 diag.snippet(os, msg.loc, msg.tag);
176 for (const auto& note : msg.notes)
177 diag.note(os, note.loc, note.str);
178 }
179
180 diag.summary(os, e.num_errors(), e.num_warnings(), e.truncated_);
181 return os;
182 }

190 void msg_(Loc loc, Tag tag, const std::function<std::string()>& fmt) {
191 assert(tag != Tag::N && "a note belongs to Error::note");
192 const auto& d = diag();
193 if (tag == Tag::Warn && d.werror) tag = Tag::Error;
194
195 if (tag == Tag::Error && d.max_errors != 0 && num_errors() >= d.max_errors) {
196 truncated_ = dropped_ = true;
197 return;
198 }
199
200 dropped_ = false;
201 ++num_[size_t(tag)];
202 msgs_.emplace_back(loc, tag, d.render(fmt));
203 }
204
205 void note_(Loc loc, const std::function<std::string()>& fmt) {
206 if (dropped_) return;
207 assert(!msgs_.empty() && "a note needs an error or warning to attach to");
208 ++num_[size_t(Tag::N)];
209 msgs_.back().notes.emplace_back(loc, diag().render(fmt));
210 }
