26/// A bijection between @p L and @p R, kept as one `ankerl::unordered_dense::map` per direction.
27/// Look up through left() and right(); a transparent hasher and equality enable heterogeneous lookup there as well.
28/// Iterating a BiMap yields its `std::pair<L, R>`s in left() order.
29/// @note Every pair is stored once per direction, so keep @p L and @p R cheap to copy - a Sym or a pointer, say.
30/// Otherwise, or if you need a pair's address to survive an insert, use BiNodeMap.
31template<class L,
32 class R,
33 class LHash = ankerl::unordered_dense::hash<L>,
34 class LEq = std::equal_to<L>,
35 class RHash = ankerl::unordered_dense::hash<R>,
36 class REq = std::equal_to<R>>
37class BiMap {
38public:
39 using Left = ankerl::unordered_dense::map<L, R, LHash, LEq>;
40 using Right = ankerl::unordered_dense::map<R, L, RHash, REq>;
41
42 /// @name Getters
43 ///@{
44 [[nodiscard]] const Left& left() const noexcept { return l2r_; }
45 [[nodiscard]] const Right& right() const noexcept { return r2l_; }
46 [[nodiscard]] size_t size() const noexcept { return l2r_.size(); }
47 [[nodiscard]] bool empty() const noexcept { return l2r_.empty(); }
48 [[nodiscard]] auto begin() const noexcept { return l2r_.begin(); }
49 [[nodiscard]] auto end() const noexcept { return l2r_.end(); }
50 ///@}
51
52 /// @name Lookup
53 /// at_left() and at_right() throw `std::out_of_range` if the key is unbound.
54 ///@{
55 [[nodiscard]] bool contains_left(const auto& l) const { return l2r_.contains(l); }
56 [[nodiscard]] bool contains_right(const auto& r) const { return r2l_.contains(r); }
57 [[nodiscard]] const R& at_left(const auto& l) const { return l2r_.at(l); }
58 [[nodiscard]] const L& at_right(const auto& r) const { return r2l_.at(r); }
59 ///@}

62 /// They take their arguments by value, so a reference into this BiMap cannot dangle halfway through.

120/// A bijection between @p L and @p R that stores every pair once, in an `ankerl::unordered_dense::segmented_map`.
121/// The right side is a set of pointers to those pairs, hashed and compared by their @p R.
122/// An insert never moves a pair, and neither does moving the BiNodeMap itself.
123/// An erase, however, moves the last pair into the freed slot, as it does in any `ankerl::unordered_dense` map.
124/// Its interface is that of BiMap, except that right() has no map to yield; find_right() takes its place.
125template<class L,
126 class R,
127 class LHash = ankerl::unordered_dense::hash<L>,
128 class LEq = std::equal_to<L>,
129 class RHash = ankerl::unordered_dense::hash<R>,
130 class REq = std::equal_to<R>>
131class BiNodeMap {
132public:
133 using Left = ankerl::unordered_dense::segmented_map<L, R, LHash, LEq>;
134 using value_type = typename Left::value_type;
135
136private:
137 using Ptr = const value_type*;
138

142
143 size_t operator()(Ptr p) const { return hash(p->second); }
144 template<detail::NotConvertibleTo<Ptr> K>
145 size_t operator()(const K& r) const {
146 return hash(r);
147 }
148
149 RHash hash;
150 };
