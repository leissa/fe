27/// ```
28/// @ref fe::snippet draws one, @ref fe::Src supplies the lines to draw.

32/// Streams @p line, then a caret run underlining the columns @p loc spans.
33/// Both sit behind a gutter of width @p gutter holding the row number.
34/// Does nothing if @p line is empty or @p loc has no column, since then there is nothing to point at.
35/// A @p loc spanning several rows underlines the remainder of its first one.
36inline std::ostream&
37snippet(std::ostream& os, Loc loc, std::string_view line, term::FG caret = term::FG::Red, int gutter = 5) {
38 // A col of 0 is Pos's "unknown" sentinel.
39 if (line.empty() || loc.begin.col == 0) return os;
40
41 auto len = utf8::num_code_points(line);
42 auto begin = size_t(loc.begin.col) - 1;
43 if (begin >= len) return os;
44
45 auto finis = loc.finis.row != loc.begin.row || loc.finis.col == 0 ? len : size_t(loc.finis.col);
46 finis = std::min(std::max(finis, begin + 1), len);
47
48 os << term::FG::Gray << std::format("{:>{}} | ", loc.begin.row, gutter) << term::FG::Reset << line << '\n'
49 << term::FG::Gray << std::format("{:>{}} | ", "", gutter) << caret;
50
51 // One blank per *code point*, since that is what Pos::col counts;
52 // a tab is echoed rather than blanked, or the carets drift by its width.
53 for (size_t i = 0, c = 0; c != begin && i < line.size(); ++c) {
54 auto n = utf8::num_bytes(char8_t(line[i]));
55 os << (line[i] == '\t' ? '\t' : ' ');
56 i += n == 0 ? 1 : n;
57 }
58 for (size_t i = begin; i != finis; ++i)
59 os << '^';
60
61 return os << term::FG::Reset << '\n';
62}

64/// The source lines a diagnostic refers to, read on demand and memoized per file.
65/// @warning Loc::path is borrowed, so a Src must not outlive the paths handed to it.
66/// Keep one per rendering rather than one for the process.
67class Src {
68public:
69 /// The line Loc::begin points at, or empty if the file or that row is unavailable.
70 std::string_view line(Loc loc) {
71 if (!loc || !loc.path) return {};
72
73 auto [i, ins] = path2lines_.try_emplace(loc.path);
74 if (ins) {
75 auto ifs = std::ifstream(*loc.path);
76 for (std::string line; std::getline(ifs, line);) {
77 if (!line.empty() && line.back() == '\r') line.pop_back();
78 i->second.emplace_back(std::move(line));
79 }
80 }
81
82 auto row = loc.begin.row;
83 return row <= i->second.size() ? std::string_view(i->second[row - 1]) : std::string_view();
84 }
