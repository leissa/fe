15/// The blueprint for a lexer with a buffer of @p K tokens to peek into the future (Lexer::ahead).
16/// You can "override" Lexer::next via CRTP (@p S is the child).
17/// The whole source has to sit in @p buf: a Pos is an index into it, so there is nothing left to
18/// keep track of - Lexer::next just hands out the byte range the code point it consumed occupied.

56 /// @name Accept
57 /// Accept next character in Lexer::buf_, depending on some condition.
58 ///@{
59 /// What should happen to the accepted char?
60 /// Normalize identifiers via Append::Lower or Append::Upper for case-insensitive languages like FORTRAN or SQL.
61 enum class Append {
62 Off, ///< Do not append accepted char to Lexer::str_.
63 On, ///< Append accepted char as is to Lexer::str_.
64 Lower, ///< Append accepted char via fe::utf8::tolower` to Lexer::str_.
65 Upper, ///< Append accepted char via fe::utf8::toupper` to Lexer::str_.
66 };
67
68 /// @returns `true` if @p pred holds.
69 /// In this case invoke Lexer::next() and append to Lexer::str_, if @p append.
70 template<Append append = Append::On, class Pred>
71 bool accept(Pred pred) {
72 if (pred(ahead())) {
73 auto c = self().next();
74 if constexpr (append != Append::Off) {
75 if constexpr (append == Append::Lower) c = fe::utf8::tolower(c);
76 if constexpr (append == Append::Upper) c = fe::utf8::toupper(c);
77 str_ += c;
78 }
79 return true;
80 }
81 return false;
82 }
