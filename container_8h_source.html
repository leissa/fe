43/// Yields pointer to element (or the element itself if it is already a pointer), if found and `nullptr` otherwise.
44/// Constness of @p container carries over to the result.
45/// @warning If the element is **not** already a pointer, this lookup will simply take the address of this element.
46/// This means that, e.g., a rehash of an `absl::flat_hash_map` will invalidate this pointer.
47template<class C, class K>
48[[nodiscard]] auto lookup(C& container, const K& key) {
49 auto i = container.find(key);
50 if constexpr (std::is_pointer_v<typename C::mapped_type>)
51 return i != container.end() ? i->second : nullptr;
52 else
53 return i != container.end() ? &i->second : nullptr;
54}
