15/// CRTP base that gives @p Self one [VLA](https://en.wikipedia.org/wiki/Variable-length_array) per element type in
16/// `Self::VLA_Types` directly behind it in memory - the C flexible array member, generalized to several arrays of
17/// different type.
18/// @p Self must be created through Arena::mk or Arena::ref, which allocate the arrays along with
19/// @p Self and fill them from the *last* `Self::VLA_Types` arguments, in that order.
20/// This base is empty - the element counts live at the front of the VLA block - so that
21/// `Self::VLA_Types` may name types only declared inside @p Self.
22/// ```
23/// class Interval : public Node, public fe::VLA<Interval> {
24/// public:
25/// using VLA_Types = std::tuple<const Expr*, const Expr*>;
26/// Interval(Loc loc, Tag from, Tag to); // no parameter for either array
27/// auto from_args() const { return vla<0>(); } // fe::View<const Expr*>
28/// auto to_args() const { return vla<1>(); }
29/// };
30///
31/// arena.ref<Interval>(loc, from, to, from_args, to_args);
32/// ```
33/// @warning Only the most derived class may carry them.
34/// A class deriving from @p Self would inherit this base and place its arrays at @p Self's offset - on top of its
35/// own members - so Arena::mk and Arena::ref reject it.
36template<class Self>
37class VLA {
38public:
39 /// The type this base was instantiated with - @p Self itself, unless a class deriving from it inherited it.
40 using VLA_Self = Self;

42 /// Element type of the @p I th VLA.
43 /// @p S defers the lookup: `Self` is still incomplete while this base is instantiated.
44 template<size_t I, class S = Self>
45 using VLA_Type = std::tuple_element_t<I, typename S::VLA_Types>;
46
47 static constexpr size_t num_vlas() noexcept { return std::tuple_size_v<typename Self::VLA_Types>; }
48

51 /// Bytes an @p Self with these element counts occupies, `sizeof(Self)` included.
52 [[nodiscard]] static constexpr size_t vla_bytes(const auto& counts) noexcept {
53 return vla_begin() + offset(num_vlas(), counts);
54 }

56 /// The @p I th VLA as a `Span<VLA_Type<I>>` - as a `View<VLA_Type<I>>`, if `*this` is `const`.
57 template<size_t I, size_t N = std::dynamic_extent, class S>
58 [[nodiscard]] auto vla(this S&& self) noexcept {
59 using T = Like<S, VLA_Type<I>>;
60 auto block = (Like<S, char>*)static_cast<Like<S, Self>*>(&self) + vla_begin();
61 auto counts = (const uint32_t*)block;
62 if constexpr (N == std::dynamic_extent)
63 return Span<T>((T*)(block + offset(I, counts)), counts[I]);
64 else
65 return Span<T, N>((T*)(block + offset(I, counts)));
66 }

84 template<size_t... Is>

86 constexpr size_t sizes[] = {sizeof(VLA_Type<Is>)...};

88

90 for (size_t i = 0; i != n; ++i)

93 }

95 /// Arena calls this once @p Self is fully constructed - only then is the downcast valid.
96 template<class... Rs>

98 static_assert(sizeof...(Rs) == num_vlas(), "one range per VLA, as the last arguments");
99
100 auto block = (char*)static_cast<Self*>(this) + vla_begin();
101 auto counts = (uint32_t*)block;
102 size_t i = 0;
103 ((counts[i++] = (uint32_t)std::ranges::size(rs)), ...);
104 fill(std::index_sequence_for<Rs...>(), block, rs...);
105 }

109 static_assert((std::same_as<std::ranges::range_value_t<Rs>, VLA_Type<Is>> && ...),

111 static_assert((std::is_trivially_destructible_v<VLA_Type<Is>> && ...),
112 "a VLA is never destroyed - the Arena reclaims it");
113
114 (std::uninitialized_copy(std::ranges::begin(rs), std::ranges::end(rs),
115 (VLA_Type<Is>*)(block + offset(Is, (const uint32_t*)block))),
116 ...);
117 }
